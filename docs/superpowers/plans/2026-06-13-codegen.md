# Codegen Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a code generator that turns a `ParsedProject` into a runnable Mastra v1 TypeScript project that `mastra build` accepts.

**Architecture:** A pure `generateProject(project, rootDir)` assembles an in-memory `FileMap` (`path → contents`) via one template-string emitter per file type; a separate `writeProject(files, outDir, rootDir, opts)` flushes the map to disk with a guarded clean. A thin CLI script wires parse → generate → write over `examples/`.

**Tech Stack:** TypeScript (ESM, NodeNext), `node:fs`/`node:path`, `tsx` for the CLI script, pnpm. No new dependencies. Tests are deferred (per `docs/SPEC.md`); each task is verified by `tsc` type-check plus eyeballing emitter output through a `tsx` one-liner, and the whole thing by a final `mastra build`.

**Spec:** [`docs/superpowers/specs/2026-06-13-codegen-design.md`](../specs/2026-06-13-codegen-design.md)

## File structure

All new code lives in `builder/src/codegen/`:

| File | Responsibility |
|---|---|
| `versions.ts` | Pinned dependency versions (from `sample-mastra/`). |
| `types.ts` | `FileMap` type + `MARKER_FILE` constant. |
| `emit-helpers.ts` | `backtickString()` — escape a string for a template literal. |
| `emit-agent.ts` | `emitAgent(agent)` → `src/mastra/agents/<id>.ts`. |
| `emit-mastra.ts` | `emitIndex(project)` → `src/mastra/index.ts`. |
| `emit-project-files.ts` | `emitPackageJson`, `emitTsconfig`, `emitGitignore`. |
| `generate.ts` | `generateProject(project, rootDir)` → `FileMap` (copies tool sources). |
| `write.ts` | `writeProject(files, outDir, rootDir, opts)` — guarded disk write. |

Modified: `builder/src/index.ts` (exports), `builder/package.json` (script), repo `.gitignore`.
New: `builder/scripts/generate-example.ts` (CLI).

**Note on `writeProject` signature:** the spec sketched `writeProject(map, outDir, opts)`, but the overlap guard needs the input root, so the real signature is `writeProject(files, outDir, rootDir, opts)`.

```ts
/** Escape a string for safe inclusion inside a `backtick` template literal:
 *  backslash, backtick, and the ${ interpolation opener. */
export function backtickString(value: string): string {
  return value
    .replace(/\\/g, '\\\\')
    .replace(/`/g, '\\`')
    .replace(/\$\{/g, '\\${');
}
```

### Task 3: Agent file emitter

**Files:**
- Create: `builder/src/codegen/emit-agent.ts`

- [ ] **Step 1: Write `emit-agent.ts`**

```ts
import type { ResolvedAgent } from '../types.js';
import { toExportName } from '../naming.js';
import { backtickString } from './emit-helpers.js';

/** Generate the source for src/mastra/agents/<id>.ts. */
export function emitAgent(agent: ResolvedAgent): string {
  const lines: string[] = [];
  lines.push(`import { Agent } from '@mastra/core/agent';`);

  // Tool imports — deduped, in listed order.
  const seenImport = new Set<string>();
  for (const tool of agent.tools) {
    if (seenImport.has(tool.exportName)) continue;
    seenImport.add(tool.exportName);
    lines.push(`import { ${tool.exportName} } from '../tools/${tool.id}';`);
  }
  lines.push('');

  const fields: string[] = [];
  fields.push(`  id: '${agent.id}',`);
  fields.push(`  name: ${JSON.stringify(agent.name)},`);
  if (agent.description) {
    fields.push(`  description: ${JSON.stringify(agent.description)},`);
  }
  fields.push(`  instructions: \`${backtickString(agent.instructions)}\`,`);
  fields.push(`  model: '${agent.model.routerString}',`);

  const settings: string[] = [];
  if (agent.model.temperature !== undefined) {
    settings.push(`temperature: ${agent.model.temperature}`);
  }
  if (agent.model.maxTokens !== undefined) {
    settings.push(`maxTokens: ${agent.model.maxTokens}`);
  }
  if (settings.length > 0) {
    fields.push(`  modelSettings: { ${settings.join(', ')} },`);
  }

  if (agent.tools.length > 0) {
    const toolVars = [...new Set(agent.tools.map((t) => t.exportName))].join(', ');
    fields.push(`  tools: { ${toolVars} },`);
  }

  lines.push(`export const ${toExportName(agent.id)} = new Agent({`);
  lines.push(...fields);
  lines.push(`});`);
  lines.push('');
  return lines.join('\n');
}
```

```ts
import type { ParsedProject } from '../types.js';
import { toExportName } from '../naming.js';

/** Generate the source for src/mastra/index.ts. */
export function emitIndex(project: ParsedProject): string {
  const lines: string[] = [];
  lines.push(`import { Mastra } from '@mastra/core/mastra';`);
  lines.push(`import { PinoLogger } from '@mastra/loggers';`);
  if (project.storage) {
    lines.push(`import { LibSQLStore } from '@mastra/libsql';`);
  }
  for (const agent of project.agents) {
    lines.push(`import { ${toExportName(agent.id)} } from './agents/${agent.id}';`);
  }
  lines.push('');

  const agentVars = project.agents.map((a) => toExportName(a.id)).join(', ');
  lines.push(`export const mastra = new Mastra({`);
  lines.push(`  agents: { ${agentVars} },`);
  if (project.storage) {
    lines.push(
      `  storage: new LibSQLStore({ id: 'mastra-storage', url: ${JSON.stringify(project.storage.url)} }),`,
    );
  }
  lines.push(`  logger: new PinoLogger({ name: 'Mastra', level: '${project.logger.level}' }),`);
  lines.push(`});`);
  lines.push('');
  return lines.join('\n');
}
```

```ts
import type { ParsedProject } from '../types.js';
import { VERSIONS } from './versions.js';

export function emitPackageJson(project: ParsedProject): string {
  const dependencies = {
    '@mastra/core': VERSIONS['@mastra/core'],
    '@mastra/loggers': VERSIONS['@mastra/loggers'],
    ...(project.storage ? { '@mastra/libsql': VERSIONS['@mastra/libsql'] } : {}),
    zod: VERSIONS.zod,
  };
  const pkg = {
    name: project.name,
    version: '1.0.0',
    type: 'module',
    engines: { node: '>=22.13.0' },
    scripts: {
      dev: 'mastra dev',
      build: 'mastra build',
      start: 'mastra start',
    },
    dependencies,
    devDependencies: {
      mastra: VERSIONS.mastra,
      typescript: VERSIONS.typescript,
      '@types/node': VERSIONS['@types/node'],
    },
  };
  return JSON.stringify(pkg, null, 2) + '\n';
}

export function emitTsconfig(): string {
  const tsconfig = {
    compilerOptions: {
      target: 'ES2022',
      module: 'ES2022',
      moduleResolution: 'bundler',
      esModuleInterop: true,
      forceConsistentCasingInFileNames: true,
      strict: true,
      skipLibCheck: true,
      noEmit: true,
      outDir: 'dist',
    },
    include: ['src/**/*'],
  };
  return JSON.stringify(tsconfig, null, 2) + '\n';
}

export function emitGitignore(): string {
  return ['node_modules', 'dist', '.mastra', '.env', '*.db', '*.db-*', ''].join('\n');
}
```

- [ ] **Step 1: Write `generate.ts`**

```ts
import { readFileSync } from 'node:fs';
import { join } from 'node:path';
import type { ParsedProject } from '../types.js';
import type { FileMap } from './types.js';
import { MARKER_FILE } from './types.js';
import { emitAgent } from './emit-agent.js';
import { emitIndex } from './emit-mastra.js';
import { emitGitignore, emitPackageJson, emitTsconfig } from './emit-project-files.js';

/** Produce the full in-memory file map for a Mastra project from a ParsedProject.
 *  Tool .ts sources are copied verbatim from rootDir; otherwise pure. */
export function generateProject(project: ParsedProject, rootDir: string): FileMap {
  const files: FileMap = {};

  files['package.json'] = emitPackageJson(project);
  files['tsconfig.json'] = emitTsconfig();
  files['.gitignore'] = emitGitignore();
  files['src/mastra/index.ts'] = emitIndex(project);

  for (const agent of project.agents) {
    files[`src/mastra/agents/${agent.id}.ts`] = emitAgent(agent);
    for (const tool of agent.tools) {
      const dest = `src/mastra/tools/${tool.id}.ts`;
      if (files[dest]) continue; // copy each tool once even if shared across agents
      files[dest] = readFileSync(join(rootDir, tool.filePath), 'utf8');
    }
  }

  files[MARKER_FILE] = 'Generated by mastra-yaml-builder. Safe to delete and regenerate.\n';
  return files;
}
```

### Task 7: Guarded disk writer

**Files:**
- Create: `builder/src/codegen/write.ts`

- [ ] **Step 1: Write `write.ts`**

```ts
import { existsSync, mkdirSync, readdirSync, rmSync, writeFileSync } from 'node:fs';
import { dirname, isAbsolute, join, relative, resolve } from 'node:path';
import type { FileMap } from './types.js';
import { MARKER_FILE } from './types.js';

export interface WriteOptions {
  /** Overwrite a non-empty directory even without our marker file. */
  force?: boolean;
}

/** Write a FileMap to outDir. Guards against clobbering the input or unrelated dirs. */
export function writeProject(
  files: FileMap,
  outDir: string,
  rootDir: string,
  opts: WriteOptions = {},
): void {
  const absOut = resolve(outDir);
  const absRoot = resolve(rootDir);

  if (isSameOrNested(absOut, absRoot)) {
    throw new Error(
      `Refusing to write: output dir ${absOut} overlaps the input dir ${absRoot}.`,
    );
  }

  if (existsSync(absOut) && readdirSync(absOut).length > 0) {
    const hasMarker = existsSync(join(absOut, MARKER_FILE));
    if (!hasMarker && !opts.force) {
      throw new Error(
        `Refusing to overwrite ${absOut}: not empty and no ${MARKER_FILE} marker. Pass force to override.`,
      );
    }
    rmSync(absOut, { recursive: true, force: true });
  }

  for (const [relPath, contents] of Object.entries(files)) {
    const dest = join(absOut, relPath);
    mkdirSync(dirname(dest), { recursive: true });
    writeFileSync(dest, contents);
  }
}

/** True if a === b, or one is nested inside the other. */
function isSameOrNested(a: string, b: string): boolean {
  if (a === b) return true;
  const within = (parent: string, child: string) => {
    const rel = relative(parent, child);
    return rel !== '' && !rel.startsWith('..') && !isAbsolute(rel);
  };
  return within(a, b) || within(b, a);
}
```
